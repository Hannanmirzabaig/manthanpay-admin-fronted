import { loginAdmin, verifyLoginOtp } from "@/services/auth/authService";
import VerifyLoginOtp from "./VerifyLoginOtp";

export default function Login() {
  const router = useRouter();
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [loginRole, setLoginRole] = useState("Admin");
  const [showPassword, setShowPassword] = useState(false);
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);
  const [showOtpVerification, setShowOtpVerification] = useState(false);
  const [pendingEmail, setPendingEmail] = useState("");

  const handleLogin = async (event) => {
    event.preventDefault();
    setError("");
    setMessage("");
    setLoading(true);

    try {
      const response = await loginAdmin({
        method: "POST",
        body: {
          identifier: username.trim(),
          password,
          role: loginRole === "Super Admin" ? "SUPER_ADMIN" : "ADMIN",
        },
      });

      setPendingEmail(username.trim());
      setMessage(response?.message || "OTP sent successfully.");
      setShowOtpVerification(true);
    } catch (err) {
      setError(err.message || "Login failed. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  const handleVerifyLoginOtp = async (otp) => {
    setError("");
    setMessage("");
    setLoading(true);

    try {
      const response = await verifyLoginOtp({
        method: "POST",
        body: {
          identifier: pendingEmail,
          otp,
        },
      });

      if (!response?.accessToken) {
        throw new Error("Authentication token not received.");
      }

      localStorage.setItem("manthanpayAuthToken", response.accessToken);
      localStorage.setItem("manthanpayLoggedInUser", response.fullName || "");
      localStorage.setItem(
        "manthanpayLoggedInEmail",
        response.email || pendingEmail
      );
      localStorage.setItem("manthanpayLoggedInRole", response.role || "");
      localStorage.setItem("manthanpayUserId", String(response.id ?? response.userId ?? ""));

      router.push("/dashboard");
    } catch (err) {
      setError(err.message || "OTP verification failed.");
    } finally {
      setLoading(false);
    }
  };

  const handleBackToLogin = () => {
    setShowOtpVerification(false);
    setError("");
    setMessage("");

    event.preventDefault();
    router.push("/forgot-password");

              <span className="point point-one"></span>

              <span className="point point-five"></span>
              <span className="point point-six"></span>
            </div>

          </div>

"use client";

import { useState } from "react";

export default function VerifyLoginOtp({
  email,
  onVerify,
  onBack,
  onResend,
  isLoading = false,
  error = "",
  message = "",
}) {
  const [otp, setOtp] = useState("");

  function handleSubmit(event) {
    event.preventDefault();

    if (otp.length !== 6 || isLoading) return;

    onVerify(otp.trim());
  }

  return (
    <div className="otp-verification-content">
      <h2>Verify Login OTP</h2>

      {email && <p>OTP sent to {email}</p>}

      {error && (
        <p role="alert" className="error-message">
          {error}
        </p>
      )}

      {message && (
        <p role="status" className="success-message">
          {message}
        </p>
      )}

      <form onSubmit={handleSubmit}>
        <input
          id="login-otp"
          type="text"
          inputMode="numeric"
          autoComplete="one-time-code"
          placeholder="Enter 6-digit OTP"
          value={otp}
          onChange={(event) =>

"use client";

import { useState } from "react";
import {
  uploadAadhaarESignPdf,
  getAadhaarESignStatus,
} from "../services/aadhaarESignService";
import { validatePdf } from "../utils/pdfValidator";

export default function useAadhaarESign() {
  const [selectedFile, setSelectedFile] = useState(null);
  const [uploading, setUploading] = useState(false);
  const [uploadResponse, setUploadResponse] = useState(null);
  const [referenceId, setReferenceId] = useState("");
  const [status, setStatus] = useState(null);
  const [checkingStatus, setCheckingStatus] = useState(false);
  const [error, setError] = useState("");

  function selectFile(file) {
    setError("");
    setUploadResponse(null);
    setStatus(null);

    if (!file) {
      setSelectedFile(null);
      return;
    }

    setSelectedFile(file);
  }

  async function uploadFile() {
    setError("");
    setUploadResponse(null);
    setStatus(null);

    try {
      await validatePdf(selectedFile);

      setUploading(true);

      const response = await uploadAadhaarESignPdf(selectedFile);

      setUploadResponse(response);

      const returnedReferenceId =
        response?.referenceId ||
        response?.data?.referenceId ||
        "";

      if (returnedReferenceId) {
        setReferenceId(String(returnedReferenceId));
      }

      return response;
    } catch (err) {
      setError(err.message || "PDF upload failed.");
      throw err;
    } finally {
      setUploading(false);
    }
  }

  async function checkStatus(value = referenceId) {
    setError("");
    setStatus(null);

    if (!value?.trim()) {
      setError("Please enter a valid Reference ID.");
      return;
    }

    try {
      setCheckingStatus(true);

      const response = await getAadhaarESignStatus(value);

      setStatus(response);

      return response;
    } catch (err) {
      setError(err.message || "Unable to check eSign status.");
      throw err;
    } finally {
      setCheckingStatus(false);
    }
  }

  function reset() {
    setSelectedFile(null);
    setUploadResponse(null);
    setReferenceId("");
    setStatus(null);
    setError("");
  }

  return {
    selectedFile,
    uploading,
    uploadResponse,
    referenceId,
    status,
    checkingStatus,
    error,
    setReferenceId,
    selectFile,
    uploadFile,

const MAX_FILE_SIZE = 10 * 1024 * 1024;

export async function validatePdf(file) {
  if (!file) {
    throw new Error("Please select a PDF file.");
  }

  if (!file.name.toLowerCase().endsWith(".pdf")) {
    throw new Error("Only PDF files are allowed.");
  }

  if (file.size > MAX_FILE_SIZE) {
    throw new Error("PDF size must not exceed 10 MB.");
  }

  if (file.size === 0) {
    throw new Error("The selected PDF is empty.");
  }

  const header = await file.slice(0, 5).arrayBuffer();
  const signature = new TextDecoder().decode(header);

  if (signature !== "%PDF-") {
    throw new Error("Invalid PDF file. Please select a valid PDF.");
  }

  return true;
}

export function formatFileSize(bytes) {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(2)} KB`;

  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
}

"use client";

import { useEffect, useState } from "react";
import { updateKycRemark } from "../services/kycService";

export default function KycRemarkEditor({ recordId, remarks }) {
  const initialRemark = remarks ?? "";

  const [savedRemark, setSavedRemark] = useState(initialRemark);
  const [draft, setDraft] = useState(initialRemark);
  const [editing, setEditing] = useState(false);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    const nextRemark = remarks ?? "";
    setSavedRemark(nextRemark);
    setDraft(nextRemark);
  }, [remarks]);

  function startEditing() {
    setDraft(savedRemark);
    setError("");
    setEditing(true);
  }

  function cancelEditing() {
    setDraft(savedRemark);
    setError("");
    setEditing(false);
  }

  async function saveRemark() {
    if (draft.length > 100) {
      setError("Remark cannot exceed 100 characters.");
      return;
    }

    setSaving(true);
    setError("");

    try {
      const result = await updateKycRemark(recordId, draft);
      const updatedRemark = result?.remarks ?? draft;

      setSavedRemark(updatedRemark);
      setDraft(updatedRemark);
      setEditing(false);
    } catch (err) {
      setError(err?.message || "Unable to save remark. Please try again.");
    } finally {
      setSaving(false);
    }
  }

  if (!editing) {
    return (
      <div className="flex min-w-[180px] max-w-[260px] items-start gap-2">
        <span className="min-w-0 flex-1 whitespace-normal break-words text-slate-700">
          {savedRemark || "No remark"}
        </span>

        <button
          type="button"
          onClick={startEditing}
          className="shrink-0 rounded-md border border-blue-200 px-2 py-1 text-xs font-medium text-blue-700 hover:bg-blue-50"
        >
          {savedRemark ? "Edit" : "Add Remark"}
        </button>
      </div>
    );
  }

  return (
    <div className="flex min-w-[220px] max-w-[280px] flex-col gap-2">
      <textarea
        value={draft}
        onChange={(event) => setDraft(event.target.value.slice(0, 100))}
        maxLength={100}
        rows={3}
        disabled={saving}
        placeholder="Enter remark (maximum 100 characters)"
        className="w-full resize-y rounded-md border border-slate-300 px-2 py-2 text-sm text-slate-800 outline-none focus:border-blue-500 focus:ring-1 focus:ring-blue-500 disabled:bg-slate-100"
      />

      <div className="flex items-center justify-between gap-2">
        <span className="text-xs text-slate-500">
          {draft.length}/100
        </span>

        <div className="flex gap-2">
          <button
            type="button"
            onClick={cancelEditing}
            disabled={saving}
            className="rounded-md border border-slate-300 px-2 py-1 text-xs text-slate-700 hover:bg-slate-100 disabled:opacity-50"
          >
            Cancel
          </button>

          <button
            type="button"
            onClick={saveRemark}
            disabled={saving}
            className="rounded-md bg-blue-600 px-3 py-1 text-xs font-medium text-white hover:bg-blue-700 disabled:opacity-50"
          >
            {saving ? "Saving..." : "Save"}
          </button>
        </div>
      </div>

      {error && (
        <p className="text-xs text-red-600" role="alert">
          {error}
        </p>
      )}
    </div>
  );
}

// Detect PDF or common image formats using MIME type and file signature.
async function detectDocumentType(blob) {
  if (!blob) return "";

  const declaredType = (blob.type || "")
    .split(";")[0]
    .trim()
    .toLowerCase();

  if (
    declaredType === "application/pdf" ||
    declaredType.startsWith("image/")
  ) {
    return declaredType;
  }

  const bytes = new Uint8Array(
    await blob.slice(0, 12).arrayBuffer()
  );

  // PDF: %PDF-
  if (
    bytes[0] === 0x25 &&
    bytes[1] === 0x50 &&
    bytes[2] === 0x44 &&
    bytes[3] === 0x46 &&
    bytes[4] === 0x2D
  ) {
    return "application/pdf";
  }

  // PNG
  if (
    bytes[0] === 0x89 &&
    bytes[1] === 0x50 &&
    bytes[2] === 0x4E &&
    bytes[3] === 0x47
  ) {
    return "image/png";
  }

  // JPEG
  if (
    bytes[0] === 0xFF &&
    bytes[1] === 0xD8 &&
    bytes[2] === 0xFF
  ) {
    return "image/jpeg";
  }

  // GIF
  if (
    bytes[0] === 0x47 &&
    bytes[1] === 0x49 &&
    bytes[2] === 0x46
  ) {
    return "image/gif";
  }

  // WEBP
  if (
    bytes[0] === 0x52 &&
    bytes[1] === 0x49 &&
    bytes[2] === 0x46 &&
    bytes[3] === 0x46 &&
    bytes[8] === 0x57 &&
    bytes[9] === 0x45 &&
    bytes[10] === 0x42 &&
    bytes[11] === 0x50
  ) {
    return "image/webp";
  }

  return "";
}

        retailerId,
        statusDocumentNames[endpoint],
        "REJECTED"
      );

      // Update UI only after both API calls succeed.
      setReviewStatus("rejected");
      window.dispatchEvent(new Event("kyc-image-status-updated"));
      handleClose();
    } catch (err) {
      setRejectError(
        err?.message ||
          "Unable to save rejection. Please try again."

    } finally {
