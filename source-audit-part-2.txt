"use client";

import { useState } from "react";
import {
  uploadAadhaarESignPdf,
  getAadhaarESignStatus,
} from "../services/aadhaarESignService";
import { validatePdf } from "../utils/pdfValidator";

export default function useAadhaarESign() {
  const [selectedFile, setSelectedFile] = useState(null);
  const [uploading, setUploading] = useState(false);
  const [uploadResponse, setUploadResponse] = useState(null);
  const [referenceId, setReferenceId] = useState("");
  const [status, setStatus] = useState(null);
  const [checkingStatus, setCheckingStatus] = useState(false);
  const [error, setError] = useState("");

  function selectFile(file) {
    setError("");
    setUploadResponse(null);
    setStatus(null);

    if (!file) {
      setSelectedFile(null);
      return;
    }

    setSelectedFile(file);
  }

  async function uploadFile() {
    setError("");
    setUploadResponse(null);
    setStatus(null);

    try {
      await validatePdf(selectedFile);

      setUploading(true);

      const response = await uploadAadhaarESignPdf(selectedFile);

      setUploadResponse(response);

      const returnedReferenceId =
        response?.referenceId ||
        response?.data?.referenceId ||
        "";

      if (returnedReferenceId) {
        setReferenceId(String(returnedReferenceId));
      }

      return response;
    } catch (err) {
      setError(err.message || "PDF upload failed.");
      throw err;
    } finally {
      setUploading(false);
    }
  }

  async function checkStatus(value = referenceId) {
    setError("");
    setStatus(null);

    if (!value?.trim()) {
      setError("Please enter a valid Reference ID.");
      return;
    }

    try {
      setCheckingStatus(true);

      const response = await getAadhaarESignStatus(value);

      setStatus(response);

      return response;
    } catch (err) {
      setError(err.message || "Unable to check eSign status.");
      throw err;
    } finally {
      setCheckingStatus(false);
    }
  }

  function reset() {
    setSelectedFile(null);
    setUploadResponse(null);
    setReferenceId("");
    setStatus(null);
    setError("");
  }

  return {
    selectedFile,
    uploading,
    uploadResponse,
    referenceId,
    status,
    checkingStatus,
    error,
    setReferenceId,
    selectFile,
    uploadFile,
    checkStatus,
    reset,
  };
}

  return apiRequest("/api/kyc/aadhaar-esign/upload", {
    method: "POST",
    body: formData,
  });
}

export async function getAadhaarESignStatus(referenceId) {
  if (!referenceId || !referenceId.trim()) {
    throw new Error("Reference ID is required.");
  }

  return apiRequest(
    `/api/kyc/aadhaar-esign/status?referenceId=${encodeURIComponent(referenceId.trim())}`,
    {
      method: "GET",
    }
  );
}

const MAX_FILE_SIZE = 10 * 1024 * 1024;

export async function validatePdf(file) {
  if (!file) {
    throw new Error("Please select a PDF file.");
  }

  if (!file.name.toLowerCase().endsWith(".pdf")) {
    throw new Error("Only PDF files are allowed.");
  }

  if (file.size > MAX_FILE_SIZE) {
    throw new Error("PDF size must not exceed 10 MB.");
  }

  if (file.size === 0) {
    throw new Error("The selected PDF is empty.");
  }

  const header = await file.slice(0, 5).arrayBuffer();
  const signature = new TextDecoder().decode(header);

  if (signature !== "%PDF-") {
    throw new Error("Invalid PDF file. Please select a valid PDF.");
  }

  return true;
}

export function formatFileSize(bytes) {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(2)} KB`;

  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
}

"use client";

import { useMemo, useState } from "react";
import { useKycRecords } from "../hooks/useKyc";
import KycTable from "./KycTable";

export default function KycList() {
  const { records, loading, error, reload } = useKycRecords();
  const [search, setSearch] = useState("");

  const filteredRecords = useMemo(() => {
    const query = search.trim().toLowerCase();

    if (!query) return records;

    return records.filter((record) =>
      [
        record.retailerId,
        record.retailerName,
        record.mobileNo,
        record.panNo,
      ].some((value) => String(value ?? "").toLowerCase().includes(query))
    );
  }, [records, search]);

  return (
    <section className="w-full min-w-0 space-y-5">
      <div>
        <h1 className="text-2xl font-bold tracking-tight text-slate-900">
          KYC Management
        </h1>
        {/* <p className="mt-1 text-sm text-slate-500">
          View and search retailer KYC records.
        </p> */}
      </div>

      <div className="flex w-full min-w-0 flex-col gap-3 sm:flex-row">
        <div className="relative min-w-0 flex-1">
          <input
            type="search"
            value={search}
            onChange={(event) => setSearch(event.target.value)}
            placeholder="Search Retailer ID, Name, Mobile, PAN..."
            className="h-11 w-full rounded-lg border border-slate-300 bg-white px-4 text-sm text-slate-800 outline-none transition placeholder:text-slate-400 focus:border-blue-700 focus:ring-2 focus:ring-blue-100"
          />
        </div>

        <button
          type="button"
          onClick={reload}
          disabled={loading}
          className="h-11 rounded-lg border border-slate-300 bg-white px-5 text-sm font-semibold text-slate-700 transition hover:bg-slate-50 disabled:cursor-not-allowed disabled:opacity-50"
        >
          {loading ? "Loading..." : "Refresh"}

            onClick={reload}
            className="text-sm font-semibold text-red-800 underline"

"use client";

import { useEffect, useState } from "react";
import { updateKycRemark } from "../services/kycService";

export default function KycRemarkEditor({ recordId, remarks }) {
  const initialRemark = remarks ?? "";

  const [savedRemark, setSavedRemark] = useState(initialRemark);
  const [draft, setDraft] = useState(initialRemark);
  const [editing, setEditing] = useState(false);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    const nextRemark = remarks ?? "";
    setSavedRemark(nextRemark);
    setDraft(nextRemark);
  }, [remarks]);

  function startEditing() {
    setDraft(savedRemark);
    setError("");
    setEditing(true);
  }

  function cancelEditing() {
    setDraft(savedRemark);
    setError("");
    setEditing(false);
  }

  async function saveRemark() {
    if (draft.length > 100) {
      setError("Remark cannot exceed 100 characters.");
      return;
    }

    setSaving(true);
    setError("");

    try {
      const result = await updateKycRemark(recordId, draft);
      const updatedRemark = result?.remarks ?? draft;

      setSavedRemark(updatedRemark);
      setDraft(updatedRemark);
      setEditing(false);
    } catch (err) {
      setError(err?.message || "Unable to save remark. Please try again.");
    } finally {
      setSaving(false);
    }
  }

  if (!editing) {
    return (
      <div className="flex min-w-[180px] max-w-[260px] items-start gap-2">
        <span className="min-w-0 flex-1 whitespace-normal break-words text-slate-700">
          {savedRemark || "No remark"}
        </span>

        <button
          type="button"
          onClick={startEditing}
          className="shrink-0 rounded-md border border-blue-200 px-2 py-1 text-xs font-medium text-blue-700 hover:bg-blue-50"
        >
          {savedRemark ? "Edit" : "Add Remark"}
        </button>
      </div>
    );
  }

  return (
    <div className="flex min-w-[220px] max-w-[280px] flex-col gap-2">
      <textarea
        value={draft}
        onChange={(event) => setDraft(event.target.value.slice(0, 100))}
        maxLength={100}
        rows={3}
        disabled={saving}
        placeholder="Enter remark (maximum 100 characters)"
        className="w-full resize-y rounded-md border border-slate-300 px-2 py-2 text-sm text-slate-800 outline-none focus:border-blue-500 focus:ring-1 focus:ring-blue-500 disabled:bg-slate-100"
      />

      <div className="flex items-center justify-between gap-2">
        <span className="text-xs text-slate-500">
          {draft.length}/100
        </span>

        <div className="flex gap-2">
          <button
            type="button"
            onClick={cancelEditing}
            disabled={saving}
            className="rounded-md border border-slate-300 px-2 py-1 text-xs text-slate-700 hover:bg-slate-100 disabled:opacity-50"
          >
            Cancel
          </button>

          <button
            type="button"
            onClick={saveRemark}
            disabled={saving}
            className="rounded-md bg-blue-600 px-3 py-1 text-xs font-medium text-white hover:bg-blue-700 disabled:opacity-50"
          >
            {saving ? "Saving..." : "Save"}
          </button>
        </div>
      </div>

      {error && (
        <p className="text-xs text-red-600" role="alert">
          {error}
        </p>
      )}
    </div>
  );
}
